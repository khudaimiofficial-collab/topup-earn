import admin from "firebase-admin"; if (!admin.apps.length) { try { if (process.env.FIREBASE_SERVICE_ACCOUNT) { admin.initializeApp({ credential: admin.credential.cert(JSON.parse(process.env.FIREBASE_SERVICE_ACCOUNT)) }); } else { admin.initializeApp({ credential: admin.credential.applicationDefault() }); } } catch (e) { console.error("Firebase init error:", e); } } const db = admin.firestore(); export default async function handler(req, res) { if (req.method !== "POST") { return res.status(405).json({ error: "Method not allowed" }); } const { telegram_id, first_name, username, referrer_id, add_points } = req.body; if (!telegram_id) { return res.status(400).json({ error: "telegram_id is required" }); } const botToken = process.env.TELEGRAM_BOT_TOKEN || ""; let photo_url = null; let bot_name = "Free Gram Token"; // 1. Fetch User Photo & Bot Name if (botToken) { try { const botRes = await fetch(`https://api.telegram.org/bot${botToken}/getMe`); const botData = await botRes.json(); if (botData.ok && botData.result.first_name) { bot_name = botData.result.first_name; } const photoRes = await fetch( `https://api.telegram.org/bot${botToken}/getUserProfilePhotos?user_id=${telegram_id}&limit=1` ); const photoData = await photoRes.json(); if (photoData.ok && photoData.result.total_count > 0) { const fileId = photoData.result.photos[0][0].file_id; const fileRes = await fetch(`https://api.telegram.org/bot${botToken}/getFile?file_id=${fileId}`); const fileData = await fileRes.json(); if (fileData.ok && fileData.result.file_path) { photo_url = `https://api.telegram.org/file/bot${botToken}/${fileData.result.file_path}`; } } } catch (err) { console.error("Telegram API fetch error:", err); } } // 2. Query User in Firestore try { const todayStr = new Date().toISOString().slice(0, 10); const userRef = db.collection("users").doc(String(telegram_id)); const userDoc = await userRef.get(); let userData; if (!userDoc.exists) { userData = { telegram_id: String(telegram_id), first_name: first_name || "Friend", username: username || "", balance: 0.0, ads_watched: 0, today_ads_watched: 0, last_ad_date: todayStr, invited_count: 0, referral_earnings: 0.0, referrer_id: referrer_id || null, channel_joined: true, created_at: new Date().toISOString() }; await userRef.set(userData); // Increment referrer's count if valid if (referrer_id && String(referrer_id) !== String(telegram_id)) { const refUser = db.collection("users").doc(String(referrer_id)); await refUser.update({ invited_count: admin.firestore.FieldValue.increment(1) }).catch(() => {}); } } else { userData = userDoc.data(); // Support direct background PTS balance adjustments if (add_points && typeof add_points === "number") { await userRef.update({ balance: admin.firestore.FieldValue.increment(add_points), ads_watched: admin.firestore.FieldValue.increment(add_points > 0 ? 1 : 0) }); userData.balance = (userData.balance || 0) + add_points; } if (userData.last_ad_date !== todayStr) { await userRef.update({ today_ads_watched: 0, last_ad_date: todayStr }); userData.today_ads_watched = 0; } } // 3. FETCH REAL REFERRALS FROM FIRESTORE let invitedUsers = []; try { const friendsSnap = await db.collection("users") .where("referrer_id", "==", String(telegram_id)) .limit(20) .get(); invitedUsers = friendsSnap.docs.map(doc => { const d = doc.data(); return { id: d.telegram_id, name: d.first_name || "Telegram User", username: d.username || "", photo_url: d.username ? `https://t.me/i/userpic/320/${d.username.replace('@', '')}.jpg` : "", earned: (Number(d.ads_watched || 0) * 1.0).toFixed(2) }; }); } catch (fErr) { console.warn("Friends query note:", fErr); } return res.status(200).json({ status: "success", balance: Number(userData.balance || 0), ads_watched: Number(userData.ads_watched || 0), today_ads_watched: Number(userData.today_ads_watched || 0), invited_count: Math.max(invitedUsers.length, Number(userData.invited_count || 0)), referral_earnings: Number(userData.referral_earnings || (invitedUsers.length * 2.0)), channel_joined: userData.channel_joined !== false, bot_name: bot_name, photo_url: photo_url, invited_users: invitedUsers }); } catch (err) { console.error("Database user error:", err); return res.status(500).json({ status: "error", message: err.message }); } }
